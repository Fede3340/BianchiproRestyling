# ==============================
# CONFIGURA_PROGETTI.ps1
# Crea automaticamente AVVIA/ONLINE/CHIUDI per:
# - Desktop\BianchiproRestyling (Vite)
# - Desktop\tuttoinsieme (Nuxt + Laravel + Caddy)
# ==============================

$ErrorActionPreference = "Stop"

function Write-Utf8NoBom([string]$Path, [string]$Text) {
  $enc = New-Object System.Text.UTF8Encoding($false)
  [System.IO.File]::WriteAllText($Path, $Text, $enc)
}

function Save-Json([string]$Path, [object]$Obj) {
  ($Obj | ConvertTo-Json -Depth 50) | Set-Content -Path $Path -Encoding UTF8
}

function Find-NuxtFrontend([string]$Root) {
  $dirs = Get-ChildItem -Path $Root -Directory -ErrorAction SilentlyContinue
  foreach($d in $dirs) {
    $hasPkg = Test-Path (Join-Path $d.FullName "package.json")
    $hasNuxt =
      (Test-Path (Join-Path $d.FullName "nuxt.config.ts")) -or
      (Test-Path (Join-Path $d.FullName "nuxt.config.js")) -or
      (Test-Path (Join-Path $d.FullName "nuxt.config.mjs"))
    if($hasPkg -and $hasNuxt) { return $d.FullName }
  }
  return $null
}

function Find-LaravelBackend([string]$Root) {
  $dirs = Get-ChildItem -Path $Root -Directory -ErrorAction SilentlyContinue
  foreach($d in $dirs) {
    if(Test-Path (Join-Path $d.FullName "artisan")) { return $d.FullName }
  }
  return $null
}

function Install-Automation([string]$ProjectRoot, [string]$Type, [string]$FrontDir, [string]$BackDir, [int]$FrontPort, [int]$BackPort, [int]$ProxyPort) {
  if(!(Test-Path $ProjectRoot)) { return }

  $portsFile = Join-Path $ProjectRoot "_PORTS.json"
  $stateFile = Join-Path $ProjectRoot "_STATO.json"
  $urlOnline = Join-Path $ProjectRoot "URL_ONLINE.txt"

  $portsObj = [pscustomobject]@{
    front = $FrontPort
    back  = $BackPort
    proxy = $ProxyPort
  }
  Save-Json $portsFile $portsObj

  # ---------------- AUTO_AVVIO_LOCALE.ps1 ----------------
  $AUTO = @'
param([switch]$NonAprireBrowser)

$ErrorActionPreference = "SilentlyContinue"

function Stop-ByPid([int]$pid) {
  if($pid -gt 0) { cmd /c "taskkill /PID $pid /T /F" | Out-Null }
}

function Stop-Port([int]$port) {
  if($port -le 0) { return }
  try {
    $conns = Get-NetTCPConnection -LocalPort $port -State Listen -ErrorAction SilentlyContinue
    foreach($c in $conns) {
      if($c.OwningProcess) { cmd /c "taskkill /PID $($c.OwningProcess) /T /F" | Out-Null }
    }
  } catch {
    $lines = cmd /c "netstat -ano | findstr :$port"
    foreach($line in $lines) {
      if($line -match "\s+LISTENING\s+(\d+)\s*$") {
        $pid = [int]$matches[1]
        cmd /c "taskkill /PID $pid /T /F" | Out-Null
      }
    }
  }
}

$root = "__ROOT__"
$state = "__STATE__"
$portsPath = Join-Path $root "_PORTS.json"
$ports = Get-Content $portsPath -Raw | ConvertFrom-Json

$frontPort = [int]$ports.front
$backPort  = [int]$ports.back
$proxyPort = [int]$ports.proxy

# Chiudere vecchie sessioni (prima per PID, poi per porta)
if(Test-Path $state) {
  try { $s = Get-Content $state -Raw | ConvertFrom-Json } catch { $s = $null }
  if($s) {
    Stop-ByPid ([int]$s.cloudflared)
    Stop-ByPid ([int]$s.caddy)
    Stop-ByPid ([int]$s.backend)
    Stop-ByPid ([int]$s.frontend)
  }
  Remove-Item $state -Force | Out-Null
}

Stop-Port $proxyPort
Stop-Port $backPort
Stop-Port $frontPort

# Permettere host trycloudflare (serve per evitare "host non consentito" su Vite)
$env:__VITE_ADDITIONAL_SERVER_ALLOWED_HOSTS = ".trycloudflare.com"

# Evitare apertura automatica di schede da strumenti vari
$env:BROWSER = "none"

$hasCaddyfile = Test-Path (Join-Path $root "Caddyfile")
$base = if($hasCaddyfile) { "http://127.0.0.1:$proxyPort" } else { "http://127.0.0.1:$frontPort" }

$frontDir = "__FRONT__"
$backDir  = "__BACK__"

# Avvio frontend
$pFront = $null
if(Test-Path $frontDir) {
  $nodeModules = Join-Path $frontDir "node_modules"
  if(!(Test-Path $nodeModules)) {
    Start-Process -FilePath "cmd.exe" -WorkingDirectory $frontDir -Wait -ArgumentList "/c","npm install"
  }

  if("__TYPE__" -eq "VITE") {
    $pFront = Start-Process -FilePath "cmd.exe" -WorkingDirectory $frontDir -PassThru `
      -ArgumentList "/k","npm run dev -- --host 127.0.0.1 --port $frontPort"
  } else {
    $pFront = Start-Process -FilePath "cmd.exe" -WorkingDirectory $frontDir -PassThru `
      -ArgumentList "/k","npx nuxi dev --host 127.0.0.1 --port $frontPort"
  }
}

# Avvio backend (solo se esiste)
$pBack = $null
if(Test-Path $backDir) {
  $pBack = Start-Process -FilePath "cmd.exe" -WorkingDirectory $backDir -PassThru `
    -ArgumentList "/k","php artisan serve --host 127.0.0.1 --port $backPort"
}

# Avvio Caddy (solo se c'è Caddyfile e caddy è installato)
$pCaddy = $null
if($hasCaddyfile) {
  $caddyCmd = Get-Command caddy -ErrorAction SilentlyContinue
  if($caddyCmd) {
    $pCaddy = Start-Process -FilePath "cmd.exe" -WorkingDirectory $root -PassThru `
      -ArgumentList "/k","caddy run --config Caddyfile --adapter caddyfile"
  } else {
    $base = "http://127.0.0.1:$frontPort"
  }
}

# Salvare stato
$obj = [ordered]@{
  frontend    = if($pFront){ $pFront.Id } else { 0 }
  backend     = if($pBack){  $pBack.Id  } else { 0 }
  caddy       = if($pCaddy){ $pCaddy.Id } else { 0 }
  cloudflared = 0
}
($obj | ConvertTo-Json) | Set-Content -Path $state -Encoding UTF8

# Attendere che risponda prima di aprire
$ok = $false
for($i=0; $i -lt 180; $i++) {
  try {
    Invoke-WebRequest -UseBasicParsing -TimeoutSec 2 -Uri $base | Out-Null
    $ok = $true
    break
  } catch {}
  Start-Sleep -Seconds 1
}

if($ok -and -not $NonAprireBrowser) {
  Start-Process $base
}

Write-Host ""
Write-Host "PRONTO (LOCALE): $base"
Write-Host "Porte: frontend=$frontPort  backend=$backPort  proxy=$proxyPort"
if($hasCaddyfile -and -not (Get-Command caddy -ErrorAction SilentlyContinue)) {
  Write-Host ""
  Write-Host "ATTENZIONE: Caddy non trovato. Il proxy non è avviato."
  Write-Host "Il sito è aperto sul frontend diretto (senza proxy)."
}
'@

  $AUTO = $AUTO.Replace("__ROOT__",  $ProjectRoot.Replace("\","\\"))
  $AUTO = $AUTO.Replace("__STATE__", $stateFile.Replace("\","\\"))
  $AUTO = $AUTO.Replace("__FRONT__", $FrontDir.Replace("\","\\"))
  $AUTO = $AUTO.Replace("__BACK__",  $BackDir.Replace("\","\\"))
  $AUTO = $AUTO.Replace("__TYPE__",  $Type)

  Write-Utf8NoBom (Join-Path $ProjectRoot "AUTO_AVVIO_LOCALE.ps1") $AUTO

  # ---------------- CONDIVIDI_ONLINE.ps1 ----------------
  $ONLINE = @'
$ErrorActionPreference = "SilentlyContinue"

$root  = "__ROOT__"
$state = "__STATE__"
$urlFile = "__URL__"

# Avviare locale senza aprire browser
& (Join-Path $root "AUTO_AVVIO_LOCALE.ps1") -NonAprireBrowser

$ports = Get-Content (Join-Path $root "_PORTS.json") -Raw | ConvertFrom-Json
$hasCaddyfile = Test-Path (Join-Path $root "Caddyfile")

$base = if($hasCaddyfile) { "http://127.0.0.1:$($ports.proxy)" } else { "http://127.0.0.1:$($ports.front)" }

$cf = Get-Command cloudflared -ErrorAction SilentlyContinue
if(-not $cf) {
  Write-Host ""
  Write-Host "ERRORE: cloudflared non trovato."
  Write-Host "Installare cloudflared e riprovare."
  exit 1
}

$out = Join-Path $root "cloudflared_out.log"
$err = Join-Path $root "cloudflared_err.log"
Remove-Item $out, $err -Force -ErrorAction SilentlyContinue | Out-Null

$p = Start-Process -FilePath $cf.Source -PassThru `
  -ArgumentList @("tunnel","--url",$base) `
  -RedirectStandardOutput $out -RedirectStandardError $err

# Salvare PID cloudflared nello stato
try {
  $s = Get-Content $state -Raw | ConvertFrom-Json
  $s.cloudflared = $p.Id
  ($s | ConvertTo-Json) | Set-Content -Path $state -Encoding UTF8
} catch {}

# Attendere URL pubblico
$pattern = "https://[a-zA-Z0-9-]+\.trycloudflare\.com"
$pub = $null

for($i=0; $i -lt 300; $i++) {
  $t = ""
  if(Test-Path $out) { $t += (Get-Content $out -Raw) }
  if(Test-Path $err) { $t += "`n" + (Get-Content $err -Raw) }
  if($t -match $pattern) { $pub = $matches[0]; break }
  Start-Sleep -Seconds 1
}

if(-not $pub) {
  Write-Host ""
  Write-Host "Non trovo il link pubblico. Aprire:"
  Write-Host " - cloudflared_out.log"
  Write-Host " - cloudflared_err.log"
  exit 1
}

Set-Content -Path $urlFile -Value $pub -Encoding ASCII
Start-Process $pub

Write-Host ""
Write-Host "PRONTO (ONLINE): $pub"
Write-Host "Link salvato in: URL_ONLINE.txt"
Write-Host "NON chiudere questa finestra: se la chiudi, l'online si spegne."
Wait-Process -Id $p.Id
'@

  $ONLINE = $ONLINE.Replace("__ROOT__",  $ProjectRoot.Replace("\","\\"))
  $ONLINE = $ONLINE.Replace("__STATE__", $stateFile.Replace("\","\\"))
  $ONLINE = $ONLINE.Replace("__URL__",   $urlOnline.Replace("\","\\"))

  Write-Utf8NoBom (Join-Path $ProjectRoot "CONDIVIDI_ONLINE.ps1") $ONLINE

  # ---------------- CHIUDI_TUTTO.ps1 ----------------
  $CLOSE = @'
$ErrorActionPreference = "SilentlyContinue"

function Stop-ByPid([int]$pid) {
  if($pid -gt 0) { cmd /c "taskkill /PID $pid /T /F" | Out-Null }
}

function Stop-Port([int]$port) {
  if($port -le 0) { return }
  try {
    $conns = Get-NetTCPConnection -LocalPort $port -State Listen -ErrorAction SilentlyContinue
    foreach($c in $conns) {
      if($c.OwningProcess) { cmd /c "taskkill /PID $($c.OwningProcess) /T /F" | Out-Null }
    }
  } catch {
    $lines = cmd /c "netstat -ano | findstr :$port"
    foreach($line in $lines) {
      if($line -match "\s+LISTENING\s+(\d+)\s*$") {
        $pid = [int]$matches[1]
        cmd /c "taskkill /PID $pid /T /F" | Out-Null
      }
    }
  }
}

$root  = "__ROOT__"
$state = "__STATE__"

$portsPath = Join-Path $root "_PORTS.json"
$ports = Get-Content $portsPath -Raw | ConvertFrom-Json
$frontPort = [int]$ports.front
$backPort  = [int]$ports.back
$proxyPort = [int]$ports.proxy

if(Test-Path $state) {
  try { $s = Get-Content $state -Raw | ConvertFrom-Json } catch { $s = $null }
  if($s) {
    Stop-ByPid ([int]$s.cloudflared)
    Stop-ByPid ([int]$s.caddy)
    Stop-ByPid ([int]$s.backend)
    Stop-ByPid ([int]$s.frontend)
  }
  Remove-Item $state -Force | Out-Null
}

Stop-Port $proxyPort
Stop-Port $backPort
Stop-Port $frontPort

Write-Host "OK: tutto chiuso."
'@

  $CLOSE = $CLOSE.Replace("__ROOT__",  $ProjectRoot.Replace("\","\\"))
  $CLOSE = $CLOSE.Replace("__STATE__", $stateFile.Replace("\","\\"))
  Write-Utf8NoBom (Join-Path $ProjectRoot "CHIUDI_TUTTO.ps1") $CLOSE

  # ---------------- BAT wrappers ----------------
  Set-Content -Path (Join-Path $ProjectRoot "AVVIA_TUTTO.bat") -Encoding ASCII -Value @(
    '@echo off',
    'powershell -NoProfile -ExecutionPolicy Bypass -File "%~dp0AUTO_AVVIO_LOCALE.ps1"',
    'if errorlevel 1 pause'
  )

  Set-Content -Path (Join-Path $ProjectRoot "CONDIVIDI_ONLINE.bat") -Encoding ASCII -Value @(
    '@echo off',
    'powershell -NoProfile -ExecutionPolicy Bypass -File "%~dp0CONDIVIDI_ONLINE.ps1"',
    'if errorlevel 1 pause'
  )

  Set-Content -Path (Join-Path $ProjectRoot "CHIUDI_TUTTO.bat") -Encoding ASCII -Value @(
    '@echo off',
    'powershell -NoProfile -ExecutionPolicy Bypass -File "%~dp0CHIUDI_TUTTO.ps1"',
    'if errorlevel 1 pause'
  )
}

# ----- Percorsi fissi sul Desktop -----
$desktop = [Environment]::GetFolderPath("Desktop")
$pathBianchi = Join-Path $desktop "BianchiproRestyling"
$pathTutto   = Join-Path $desktop "tuttoinsieme"

# ----- BianchiProRestyling (Vite in root) -----
if(Test-Path $pathBianchi) {
  $front = $pathBianchi
  Install-Automation `
    -ProjectRoot $pathBianchi `
    -Type "VITE" `
    -FrontDir $front `
    -BackDir  "" `
    -FrontPort 3000 `
    -BackPort 0 `
    -ProxyPort 0
}

# ----- tuttoinsieme (Nuxt + Laravel + Caddy) -----
if(Test-Path $pathTutto) {
  $frontDir = Find-NuxtFrontend $pathTutto
  $backDir  = Find-LaravelBackend $pathTutto

  if(-not $frontDir) { throw "Dentro Desktop\tuttoinsieme non trovo il frontend Nuxt (package.json + nuxt.config.*)" }
  if(-not $backDir)  { throw "Dentro Desktop\tuttoinsieme non trovo il backend Laravel (file artisan)" }

  Install-Automation `
    -ProjectRoot $pathTutto `
    -Type "NUXT" `
    -FrontDir $frontDir `
    -BackDir  $backDir `
    -FrontPort 3001 `
    -BackPort 8000 `
    -ProxyPort 8787
}

Write-Host ""
Write-Host "FATTO."
Write-Host "Ora dentro ogni cartella trovi:"
Write-Host " - AVVIA_TUTTO.bat"
Write-Host " - CONDIVIDI_ONLINE.bat"
Write-Host " - CHIUDI_TUTTO.bat"
Write-Host ""
Write-Host "Se una cartella non esiste sul Desktop, non viene configurata."
