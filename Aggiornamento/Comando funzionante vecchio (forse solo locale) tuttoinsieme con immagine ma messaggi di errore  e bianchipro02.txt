# === PERCORSI ===
$T_ROOT = "$env:USERPROFILE\Desktop\tuttoinsieme"
$T_BACK = Join-Path $T_ROOT "laravel-spedizionefacile-main"
$T_NUXT = Join-Path $T_ROOT "nuxt-spedizionefacile-master"

$B_ROOT = "$env:USERPROFILE\Desktop\BianchiproRestyling"

# === FUNZIONE: KILL PORTE ===
function Kill-Port([int]$port){
  $rows = netstat -aon | findstr (":$port ")
  if($rows){
    $ids = $rows | ForEach-Object { ($_ -split '\s+')[-1] } | Sort-Object -Unique
    foreach($procId in $ids){
      Stop-Process -Id $procId -Force -ErrorAction SilentlyContinue
    }
  }
}

# Spegnere tutto ciò che può creare conflitto
Kill-Port 8787
Kill-Port 8000
Kill-Port 3001
Kill-Port 3000

# === CONTROLLI MINIMI CARTELLE ===
if(!(Test-Path (Join-Path $T_BACK "artisan"))) { throw "Tuttoinsieme: Laravel non trovato in $T_BACK" }
if(!(Test-Path (Join-Path $T_NUXT "package.json"))) { throw "Tuttoinsieme: Nuxt non trovato in $T_NUXT" }
if(!(Test-Path (Join-Path $B_ROOT "package.json"))) { throw "BianchiPro: non vedo package.json in $B_ROOT (cartella sbagliata?)" }

# =========================================================
# 1) TUTTOINSIEME: FORZARE SEMPRE Nuxt su 3001 + Caddy su 8787
# =========================================================
$T_BASE = "http://127.0.0.1:8787"
Set-Content -Path (Join-Path $T_ROOT "BASE_URL.txt") -Value $T_BASE -Encoding UTF8

# Caddyfile: 8787 -> API(8000) e resto -> Nuxt(3001)
$caddy = @(
":8787 {",
"  @api path /api/* /sanctum/*",
"  reverse_proxy @api 127.0.0.1:8000",
"  reverse_proxy 127.0.0.1:3001",
"}"
)
Set-Content -Path (Join-Path $T_ROOT "Caddyfile") -Value $caddy -Encoding UTF8

$T_PID = Join-Path $T_ROOT "PIDS.json"

$T_AUTO = @"
param([switch]`$NonAprireBrowser)

`$ErrorActionPreference = "Stop"
`$root = "$T_ROOT"
`$back = "$T_BACK"
`$nuxt = "$T_NUXT"
`$base = "http://127.0.0.1:8787"
`$pidf = "$T_PID"

# Stop vecchi PID se esistono
if (Test-Path `$pidf) {
  try {
    `$p = Get-Content `$pidf -Raw | ConvertFrom-Json
    foreach (`$procId in @(`$p.backend,`$p.frontend,`$p.caddy)) {
      if (`$procId) { Stop-Process -Id `$procId -Force -ErrorAction SilentlyContinue }
    }
  } catch {}
  Remove-Item `$pidf -Force -ErrorAction SilentlyContinue
}

# Avvio (3 finestre minimizzate)
`$pBack  = Start-Process -FilePath "cmd.exe" -WorkingDirectory `$back -WindowStyle Minimized -PassThru -ArgumentList "/k","php artisan serve --host 127.0.0.1 --port 8000"
`$pNuxt  = Start-Process -FilePath "cmd.exe" -WorkingDirectory `$nuxt -WindowStyle Minimized -PassThru -ArgumentList "/k","npm run dev -- --host 127.0.0.1 --port 3001"
`$pCaddy = Start-Process -FilePath "cmd.exe" -WorkingDirectory `$root -WindowStyle Minimized -PassThru -ArgumentList "/k","caddy run --config .\Caddyfile --adapter caddyfile"

@{ backend=`$pBack.Id; frontend=`$pNuxt.Id; caddy=`$pCaddy.Id } | ConvertTo-Json | Set-Content -Path `$pidf -Encoding UTF8

Start-Sleep -Seconds 2
if (-not `$NonAprireBrowser) { Start-Process `$base }
Write-Host "TUTTOINSIEME: `$base (Nuxt=3001, API=8000)"
"@
Set-Content -Path (Join-Path $T_ROOT "AUTO_AVVIO_TOTALE.ps1") -Value $T_AUTO -Encoding UTF8

Set-Content -Path (Join-Path $T_ROOT "AVVIA_TUTTO.bat") -Encoding ASCII -Value @(
'@echo off',
'powershell -NoProfile -ExecutionPolicy Bypass -File "%~dp0AUTO_AVVIO_TOTALE.ps1"',
'if errorlevel 1 pause'
)

# =========================================================
# 2) BIANCHIPRO: FORZARE SEMPRE Nuxt su 3000 (una sola scheda)
# =========================================================
$B_BASE = "http://127.0.0.1:3000"
$B_PID  = Join-Path $B_ROOT "PIDS_BIANCHIPRO.json"

$B_AUTO = @"
param([switch]`$NonAprireBrowser)

`$ErrorActionPreference = "Stop"
`$root = "$B_ROOT"
`$base = "http://127.0.0.1:3000"
`$pidf = "$B_PID"

if (Test-Path `$pidf) {
  try {
    `$p = Get-Content `$pidf -Raw | ConvertFrom-Json
    if (`$p.frontend) { Stop-Process -Id `$p.frontend -Force -ErrorAction SilentlyContinue }
  } catch {}
  Remove-Item `$pidf -Force -ErrorAction SilentlyContinue
}

`$pNuxt = Start-Process -FilePath "cmd.exe" -WorkingDirectory `$root -WindowStyle Minimized -PassThru -ArgumentList "/k","npm run dev -- --host 127.0.0.1 --port 3000"
@{ frontend=`$pNuxt.Id } | ConvertTo-Json | Set-Content -Path `$pidf -Encoding UTF8

Start-Sleep -Seconds 2
if (-not `$NonAprireBrowser) { Start-Process `$base }
Write-Host "BIANCHIPRO: `$base (Nuxt=3000)"
"@
Set-Content -Path (Join-Path $B_ROOT "AUTO_AVVIO_BIANCHIPRO.ps1") -Value $B_AUTO -Encoding UTF8

Set-Content -Path (Join-Path $B_ROOT "AVVIA_TUTTO.bat") -Encoding ASCII -Value @(
'@echo off',
'powershell -NoProfile -ExecutionPolicy Bypass -File "%~dp0AUTO_AVVIO_BIANCHIPRO.ps1"',
'if errorlevel 1 pause'
)

Write-Host ""
Write-Host "OK. File ripristinati."
Write-Host "Ora avvia: 1) Tuttoinsieme -> $T_ROOT\AVVIA_TUTTO.bat"
Write-Host "           2) BianchiPro   -> $B_ROOT\AVVIA_TUTTO.bat"
