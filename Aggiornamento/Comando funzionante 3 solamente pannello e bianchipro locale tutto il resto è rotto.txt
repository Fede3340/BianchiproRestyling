$ErrorActionPreference = "Stop"

function C { param([string]$m,[string]$col="Cyan") Write-Host $m -ForegroundColor $col }
function Ensure-Dir { param([string]$p) if(-not (Test-Path -LiteralPath $p)) { New-Item -ItemType Directory -Path $p -Force | Out-Null } }

function Find-Desktop {
  $cands = @()
  if (Test-Path -LiteralPath "C:\Users\Feder\Desktop") { $cands += "C:\Users\Feder\Desktop" }
  $cands += [Environment]::GetFolderPath("Desktop")
  foreach($d in $cands){
    if($d -and (Test-Path -LiteralPath $d)){ return $d }
  }
  throw "Desktop non trovato."
}

function Write-AsciiFile {
  param([string]$path,[string[]]$lines)
  $txt = ($lines -join "`r`n") + "`r`n"
  Set-Content -LiteralPath $path -Value $txt -Encoding Ascii -Force
}

function Cleanup-Pannelli {
  param([string]$root)
  $patterns = @(
    "PANNELLO*.BAK*","*.BAK_*","*_BAK_*",
    "AUTO_AVVIO*.ps1","AVVIA_TUTTO.*",
    "_PANNELLO_LIB*","_COMUNE*","_PORTS.json","_STATE.json"
  )
  foreach($pat in $patterns){
    Get-ChildItem -LiteralPath $root -Filter $pat -File -ErrorAction SilentlyContinue |
      Remove-Item -Force -ErrorAction SilentlyContinue
  }
}

$desktop = Find-Desktop

$bpRoot = Join-Path $desktop "BianchiproRestyling"
$tiRoot = Join-Path $desktop "tuttoinsieme"

$targets = @(
  @{ Root = $bpRoot; Kind = "BP"; Name = "BianchiproRestyling" },
  @{ Root = $tiRoot; Kind = "TI"; Name = "tuttoinsieme" }
)

C "RIPRISTINO PANNELLI - avvio" "Cyan"
C ("Desktop usato: " + $desktop) "DarkCyan"

foreach($t in $targets){
  if(-not (Test-Path -LiteralPath $t.Root)){
    C ("ERRORE: cartella mancante: " + $t.Root) "Red"
    continue
  }

  Cleanup-Pannelli $t.Root
  Ensure-Dir (Join-Path $t.Root "_LOGS")

  $batPanel = @(
'@echo off',
'cd /d "%~dp0"',
'powershell -NoProfile -ExecutionPolicy Bypass -File "%~dp0PANNELLO.ps1"'
  )
  $batStart = @(
'@echo off',
'cd /d "%~dp0"',
'powershell -NoProfile -ExecutionPolicy Bypass -File "%~dp0PANNELLO.ps1" start',
'pause >nul'
  )
  $batShare = @(
'@echo off',
'cd /d "%~dp0"',
'powershell -NoProfile -ExecutionPolicy Bypass -File "%~dp0PANNELLO.ps1" share',
'pause >nul'
  )
  $batStop = @(
'@echo off',
'cd /d "%~dp0"',
'powershell -NoProfile -ExecutionPolicy Bypass -File "%~dp0PANNELLO.ps1" stop',
'pause >nul'
  )

  Write-AsciiFile (Join-Path $t.Root "PANNELLO.bat") $batPanel
  Write-AsciiFile (Join-Path $t.Root "AVVIA_LOCALE.bat") $batStart
  Write-AsciiFile (Join-Path $t.Root "CONDIVIDI_ONLINE.bat") $batShare
  Write-AsciiFile (Join-Path $t.Root "CHIUDI_TUTTO.bat") $batStop

  if($t.Kind -eq "BP"){
    Write-AsciiFile (Join-Path $t.Root "PANNELLO.ps1") (Get-BPPanelLines)
  } else {
    Write-AsciiFile (Join-Path $t.Root "PANNELLO.ps1") (Get-TIPanelLines)
  }

  C ("OK: pannelli ripristinati in " + $t.Root) "Green"
}
