function Write-Utf8NoBom([string]$Path,[string]$Text){
  $enc = New-Object System.Text.UTF8Encoding($false)
  [System.IO.File]::WriteAllText($Path,$Text,$enc)
}
function Write-Ascii([string]$Path,[string]$Text){
  [System.IO.File]::WriteAllText($Path,$Text,[System.Text.Encoding]::ASCII)
}
function Ensure-Dir([string]$Path){
  if(-not (Test-Path -LiteralPath $Path)){ New-Item -ItemType Directory -Force -Path $Path | Out-Null }
}
function Require-Folder([string]$Path){
  if(-not (Test-Path -LiteralPath $Path)){ throw "Cartella non trovata: $Path" }
}

function Cleanup-Pannelli([string]$root){
  $patterns = @(
    "PANNELLO*.BAK*","*.BAK_*","*_BAK_*",
    "AUTO_AVVIO*.ps1","AVVIA_TUTTO.*","AVVIA_LOCALE.*","CONDIVIDI_ONLINE.*","CHIUDI_TUTTO.*",
    "_PANNELLO_LIB*","_COMUNE*","_PORTS.json","_STATE.json","_PANNELLO_STATE.json",
    "Caddyfile.trycloudflare","Caddyfile.trycloudflare*",
    "URL_ONLINE.txt"
  )
  foreach($pat in $patterns){
    try{
      Get-ChildItem -LiteralPath $root -Filter $pat -File -ErrorAction SilentlyContinue |
        Remove-Item -Force -ErrorAction SilentlyContinue
    } catch {}
  }
}

function T([string]$m,[string]$c="Gray"){
  $ts=(Get-Date).ToString("HH:mm:ss")
  Write-Host "[$ts] $m" -ForegroundColor $c
}

function Save-Json([string]$Path,[object]$Obj){
  ($Obj | ConvertTo-Json -Depth 20) | Set-Content -Encoding UTF8 -Path $Path
}
function Load-Json([string]$Path){
  if(Test-Path $Path){
    try { return (Get-Content $Path -Raw | ConvertFrom-Json) } catch { return $null }
  }
  return $null
}

function CmdLineOfPid([int]$procId){
  try {
    $w = Get-CimInstance Win32_Process -Filter ("ProcessId=" + $procId)
    if($w -and $w.CommandLine){ return $w.CommandLine } else { return "" }
  } catch { return "" }
}
function Stop-Tree([int]$procId){
  if(-not $procId){ return }
  if($procId -le 10){ return }
  try{ cmd /c ("taskkill /PID " + $procId + " /T /F") | Out-Null } catch {}
}
function Safe-Kill-ByPort([int]$port){
  try{
    $conns = Get-NetTCPConnection -LocalPort $port -State Listen -ErrorAction SilentlyContinue
    if(-not $conns){ return }
    $pids = $conns | Select-Object -ExpandProperty OwningProcess -Unique
    foreach($p in $pids){
      if(-not $p){ continue }
      if($p -le 10){ continue }
      $cl = CmdLineOfPid $p
      if($cl -and ($cl -like ("*" + $root + "*"))){
        Stop-Tree $p
      }
    }
  } catch {}
}

function Read-State(){
  $s = Load-Json $statePath
  if(-not $s){ $s = [pscustomobject]@{} }
  return $s
}
function Write-State($s){ Save-Json $statePath $s }
function Clear-State(){ Remove-Item $statePath -Force -ErrorAction SilentlyContinue | Out-Null }

function ReadKeyNoEnter(){
  $k = $Host.UI.RawUI.ReadKey("NoEcho,IncludeKeyDown")
  return ($k.Character.ToString()).ToUpper()
}

function Find-Frontend(){
  $needAny = @("package.json","vite.config.js","vite.config.ts","vite.config.mjs","index.html")
  $rootHas = $true
  foreach($f in @("package.json","index.html")){
    if(-not (Test-Path (Join-Path $root $f))){ $rootHas = $false; break }
  }
  if($rootHas){ return $root }

  $dirs = Get-ChildItem -Path $root -Directory -ErrorAction SilentlyContinue
  foreach($d in $dirs){
    $hits = 0
    foreach($f in $needAny){
      if(Test-Path (Join-Path $d.FullName $f)){ $hits++ }
    }
    if($hits -ge 2){ return $d.FullName }
  }

  throw "Non trovo il FRONT (Vite): manca package.json+index.html (o vite.config) in root o in una sottocartella."
}

function Find-Exe([string]$name){
  $cmd = Get-Command $name -ErrorAction SilentlyContinue
  if($cmd -and $cmd.Source){ return $cmd.Source }
  $local = Join-Path $root $name
  if(Test-Path -LiteralPath $local){ return $local }
  return $null
}

function Start-CmdLogged([string]$name,[string]$command,[string]$workDir){
  $out = Join-Path $logsDir ($name + ".out.log")
  $err = Join-Path $logsDir ($name + ".err.log")
  try{ Remove-Item $out,$err -Force -ErrorAction SilentlyContinue | Out-Null } catch {}

  $args = @("/c",$command)

  $p = Start-Process -FilePath "cmd.exe" `
    -ArgumentList $args `
    -WorkingDirectory $workDir `
    -PassThru `
    -WindowStyle Hidden `
    -RedirectStandardOutput $out `
    -RedirectStandardError  $err

  return [pscustomobject]@{ id=$p.Id; out=$out; err=$err }
}

function Test-Tcp([int]$port,[int]$timeoutMs=600){
  try{
    $c = New-Object System.Net.Sockets.TcpClient
    $iar = $c.BeginConnect("127.0.0.1",$port,$null,$null)
    $ok = $iar.AsyncWaitHandle.WaitOne($timeoutMs,$false)
    if(-not $ok){ $c.Close(); return $false }
    $c.EndConnect($iar) | Out-Null
    $c.Close()
    return $true
  } catch { return $false }
}
function Wait-Port([int]$port,[int]$timeoutSec=180){
  $start = Get-Date
  $lastTick = -1
  while(((Get-Date) - $start).TotalSeconds -lt $timeoutSec){
    $sec = [int](((Get-Date) - $start).TotalSeconds)
    if($sec -ne $lastTick){
      $lastTick = $sec
      if(($sec % 5) -eq 0){ T ("Attendere: porta " + $port + " ... (" + $sec + "s)") "Yellow" }
    }
    if(Test-Tcp $port 600){ return $true }
    Start-Sleep -Milliseconds 250
  }
  return $false
}

function Wait-Http([string]$url,[int]$timeoutSec=60){
  $start = Get-Date
  while(((Get-Date) - $start).TotalSeconds -lt $timeoutSec){
    try{
      $req = [System.Net.HttpWebRequest]::Create($url)
      $req.Method = "GET"
      $req.Timeout = 2000
      $resp = $req.GetResponse()
      $code = [int]$resp.StatusCode
      $resp.Close()
      if($code -ge 200 -and $code -lt 500){ return $true }
    } catch [System.Net.WebException] {
      if($_.Exception.Response){
        try{
          $r = $_.Exception.Response
          $code = [int]$r.StatusCode
          $r.Close()
          if($code -ge 200 -and $code -lt 500){ return $true }
        } catch {}
      }
    } catch {}
    Start-Sleep -Milliseconds 500
  }
  return $false
}

function Stop-All(){
  T "Chiusura completa (solo BianchiproRestyling)..." "Yellow"

  $s = Read-State
  try{
    $procId = [int]($s.frontend)
    if($procId -gt 0){ Stop-Tree $procId }
  } catch {}

  try{
    $procId = [int]($s.cloudflared)
    if($procId -gt 0){ Stop-Tree $procId }
  } catch {}

  Safe-Kill-ByPort 3000
  Safe-Kill-ByPort 8787

  Clear-State
  T "OK: Bianchi Pro chiuso." "Green"
}

function WaitCloudUrl([string]$outPath,[string]$errPath,[int]$timeoutSec){
  $end = (Get-Date).AddSeconds($timeoutSec)
  while((Get-Date) -lt $end){
    $txt = ""
    if(Test-Path -LiteralPath $outPath){ $txt += (Get-Content -LiteralPath $outPath -Raw -ErrorAction SilentlyContinue) }
    if(Test-Path -LiteralPath $errPath){ $txt += "`n" + (Get-Content -LiteralPath $errPath -Raw -ErrorAction SilentlyContinue) }
    $m = [regex]::Match($txt,"https://[a-z0-9-]+\.trycloudflare\.com")
    if($m.Success){ return $m.Value }
    Start-Sleep -Milliseconds 400
  }
  return ""
}

function Start-Local([switch]$NonAprireBrowser,[switch]$SoloAvvio){
  Stop-All

  $frontDir = Find-Frontend
  $frontPort = 3000
  $base = "http://127.0.0.1:$frontPort/"

  T "==============================" "Cyan"
  T "BIANCHI PRO - AVVIO LOCALE" "Cyan"
  T "==============================" "Cyan"
  T ("FRONT: " + $frontDir) "DarkGray"
  T ("URL  : " + $base.TrimEnd('/')) "DarkGray"

  $env:__VITE_ADDITIONAL_SERVER_ALLOWED_HOSTS = ".trycloudflare.com"

  $cmd = 'cd /d "' + $frontDir + '" && npm run dev -- --host 127.0.0.1 --port ' + $frontPort
  $pFront = Start-CmdLogged "frontend" $cmd $frontDir

  $s = [pscustomobject]@{
    frontend    = $pFront.id
    cloudflared = 0
    base        = $base
    port        = $frontPort
  }
  Write-State $s

  if(-not (Wait-Port $frontPort 180)){
    T ("ERRORE: porta " + $frontPort + " non in ascolto.") "Red"
    T ("Aprire log: " + $logsDir) "Red"
    Start-Process $logsDir
    return
  }

  T "Porta in ascolto. Verifico risposta HTTP..." "Yellow"
  $httpOk = Wait-Http $base 60
  if($httpOk){
    T ("PRONTO (LOCALE): " + $base.TrimEnd('/')) "Green"
  } else {
    T ("PRONTO (TCP): " + $base.TrimEnd('/') + " (HTTP lento o bloccato, controllare log)") "Yellow"
  }

  if(-not $NonAprireBrowser){ Start-Process $base }

  if($SoloAvvio){ return }

  T "Comandi: O=riaprire | L=log | Q=chiudere tutto" "Yellow"
  while($true){
    if([Console]::KeyAvailable){
      $k = [Console]::ReadKey($true)
      $key = ($k.KeyChar.ToString()).ToUpper()
      if($key -eq "O"){ Start-Process $base }
      elseif($key -eq "L"){ Start-Process $logsDir }
      elseif($key -eq "Q"){ break }
    }
    Start-Sleep -Milliseconds 200
  }

  Stop-All
}

function Share-Online(){
  Start-Local -NonAprireBrowser -SoloAvvio

  $s = Read-State
  $base = $s.base
  $frontPort = 3000
  try{ $frontPort = [int]$s.port } catch {}

  if(-not $base){ $base = "http://127.0.0.1:$frontPort/" }

  T "==============================" "Cyan"
  T "BIANCHI PRO - CONDIVISIONE ONLINE" "Cyan"
  T "==============================" "Cyan"
  T ("Base locale: " + $base.TrimEnd('/')) "Gray"

  $cloudflared = Find-Exe "cloudflared.exe"
  if(-not $cloudflared){ $cloudflared = (Get-Command cloudflared -ErrorAction SilentlyContinue).Source }
  if(-not $cloudflared){ throw "cloudflared non trovato." }

  $out = Join-Path $logsDir "cloudflared.out.log"
  $err = Join-Path $logsDir "cloudflared.err.log"
  try{ Remove-Item $out,$err -Force -ErrorAction SilentlyContinue | Out-Null } catch {}

  $args = @(
    "tunnel",
    "--protocol","http2",
    "--url",$base.TrimEnd('/'),
    "--http-host-header",("localhost:" + $frontPort),
    "--no-autoupdate"
  )

  $pTun = Start-Process -FilePath $cloudflared -ArgumentList $args -PassThru -WindowStyle Hidden `
    -RedirectStandardOutput $out -RedirectStandardError $err

  $s.cloudflared = $pTun.Id
  Write-State $s

  T "Tunnel in avvio... attendere link pubblico..." "Yellow"
  $pub = WaitCloudUrl $out $err 180
  if(-not $pub){
    T "ERRORE: non trovo il link pubblico. Apro i log." "Red"
    Start-Process $logsDir
    return
  }

  $urlFile = Join-Path $root "URL_ONLINE.txt"
  Set-Content -Encoding ASCII -Path $urlFile -Value $pub

  T ("PRONTO (ONLINE): " + $pub) "Green"

  T "Verifico raggiungibilità prima di aprire..." "Yellow"
  [void](Wait-Http $pub 120)

  Start-Process $pub

  T "Comandi: O=riaprire | L=log | Q=chiudere tunnel+locale" "Yellow"
  while($true){
    if([Console]::KeyAvailable){
      $k = [Console]::ReadKey($true)
      $key = ($k.KeyChar.ToString()).ToUpper()
      if($key -eq "O"){ Start-Process $pub }
      elseif($key -eq "L"){ Start-Process $logsDir }
      elseif($key -eq "Q"){ break }
    }
    Start-Sleep -Milliseconds 200
  }

  Stop-All
}

function Menu(){
  while($true){
    Clear-Host
    T "==============================" "Cyan"
    T "BIANCHI PRO - PANNELLO" "Cyan"
    T "==============================" "Cyan"
    Write-Host ""
    Write-Host "[1] Avvia LOCALE (apre solo quando pronto)" -ForegroundColor White
    Write-Host "[2] Condividi ONLINE (apre solo link pubblico quando pronto)" -ForegroundColor White
    Write-Host "[3] Apri cartella LOG" -ForegroundColor White
    Write-Host "[4] Chiudi TUTTO" -ForegroundColor White
    Write-Host "[Q] Esci" -ForegroundColor White
    Write-Host ""
    Write-Host "Premere un tasto (senza invio)..." -ForegroundColor Yellow

    $k = ReadKeyNoEnter
    if($k -eq "1"){ Start-Local; }
    elseif($k -eq "2"){ Share-Online; }
    elseif($k -eq "3"){ Start-Process $logsDir; Start-Sleep -Milliseconds 700 }
    elseif($k -eq "4"){ Stop-All; Start-Sleep -Milliseconds 700 }
    elseif($k -eq "Q"){ break }
  }
}

  Write-Host "ERRORE: $($_.Exception.Message)" -ForegroundColor Red
  Write-Host "Premere INVIO per chiudere..." -ForegroundColor Yellow
