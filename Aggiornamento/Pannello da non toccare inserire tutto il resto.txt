# ============================================================================
# INSTALLA.ps1 - Installa e ripara i pannelli per ENTRAMBI i progetti
# Eseguire UNA SOLA VOLTA, poi usare PANNELLO.bat in ogni cartella.
# Compatibile con Windows PowerShell 5.1
# ============================================================================
$ErrorActionPreference = Continue

# ---------- Utilità ----------
function C { param([string]$m,[string]$col=Cyan) Write-Host $m -ForegroundColor $col }
function Ensure-Dir([string]$p) { if (-not (Test-Path -LiteralPath $p)) { New-Item -ItemType Directory -Path $p -Force  Out-Null } }

function Write-AsciiFile {
    param([string]$path,[string[]]$lines)
    $txt = ($lines -join `r`n) + `r`n
    [System.IO.File]WriteAllText($path, $txt, [System.Text.Encoding]ASCII)
}

function Write-Utf8File {
    param([string]$path,[string[]]$lines)
    $txt = ($lines -join `r`n) + `r`n
    # Senza BOM per evitare problemi con JSON e config
    $enc = New-Object System.Text.UTF8Encoding $false
    [System.IO.File]WriteAllText($path, $txt, $enc)
}

function Find-Desktop {
    $cands = @()
    if (Test-Path -LiteralPath CUsersFederDesktop) { $cands += CUsersFederDesktop }
    $cands += [Environment]GetFolderPath(Desktop)
    foreach ($d in $cands) {
        if ($d -and (Test-Path -LiteralPath $d)) { return $d }
    }
    throw Desktop non trovato.
}

function Cleanup-Old {
    param([string]$root)
    $patterns = @(
        .BAK_,_BAK_,AUTO_AVVIO.ps1,
        _PANNELLO_LIB,_COMUNE,_PORTS.json,_STATE.json,
        Caddyfile.trycloudflare  # ← era il Caddyfile sbagliato
    )
    foreach ($pat in $patterns) {
        Get-ChildItem -LiteralPath $root -Filter $pat -File -ErrorAction SilentlyContinue 
            Remove-Item -Force -ErrorAction SilentlyContinue
    }
}

# ============================================================================
# PATCH AUTOMATICA nuxt.config.ts
# ============================================================================
function Patch-NuxtConfig {
    param([string]$nuxtDir, [int]$port)

    $configPath = Join-Path $nuxtDir nuxt.config.ts
    if (-not (Test-Path -LiteralPath $configPath)) {
        C   WARN nuxt.config.ts non trovato in $nuxtDir Yellow
        return
    }

    $content = Get-Content -LiteralPath $configPath -Raw -Encoding UTF8

    # Controlla se già patchato
    if ($content -match trycloudflare.com) {
        C   nuxt.config.ts già patchato (skip) DarkCyan
        return
    }

    # Backup
    $backup = $configPath + .BAK_PRIMA_PATCH
    Copy-Item -LiteralPath $configPath -Destination $backup -Force

    # Inserisce devServer + vite.server subito prima della chiusura }) finale
    $patch = @

   === PATCH AUTO PANNELLO - non rimuovere ===
  devServer { port $port, host '0.0.0.0' },
  vite {
    server {
      allowedHosts ['localhost','127.0.0.1','.trycloudflare.com'],
      open false,
    }
  },
   === FINE PATCH ===

@

    # Rimuove l'ultima }) e la ricrea con il patch
    $lastBrace = $content.LastIndexOf(}))
    if ($lastBrace -gt 0) {
        $newContent = $content.Substring(0, $lastBrace) + $patch + })
        $enc = New-Object System.Text.UTF8Encoding $false
        [System.IO.File]WriteAllText($configPath, $newContent, $enc)
        C   nuxt.config.ts patchato (porta $port + allowedHosts) Green
    } else {
        C   WARN nuxt.config.ts non ha struttura riconosciuta, patch saltata Yellow
    }
}

# ============================================================================
# PATCH AUTOMATICA vite.config (BianchiPro)
# ============================================================================
function Patch-ViteConfig {
    param([string]$projDir)

    # Cerca tutti i possibili nomi del vite config
    $viteFile = $null
    foreach ($name in @(vite.config.ts,vite.config.js,vite.config.mjs)) {
        $p = Join-Path $projDir $name
        if (Test-Path -LiteralPath $p) { $viteFile = $p; break }
    }

    if (-not $viteFile) {
        C   WARN vite.config non trovato in $projDir Yellow
        return
    }

    $content = Get-Content -LiteralPath $viteFile -Raw -Encoding UTF8

    if ($content -match trycloudflare.com) {
        C   vite.config già patchato (skip) DarkCyan
        return
    }

    $backup = $viteFile + .BAK_PRIMA_PATCH
    Copy-Item -LiteralPath $viteFile -Destination $backup -Force

    $patch = @

   === PATCH AUTO PANNELLO - non rimuovere ===
  server {
    allowedHosts ['localhost','127.0.0.1','.trycloudflare.com'],
    open false,
  },
   === FINE PATCH ===

@

    $lastBrace = $content.LastIndexOf(}))
    if ($lastBrace -gt 0) {
        $newContent = $content.Substring(0, $lastBrace) + $patch + })
        $enc = New-Object System.Text.UTF8Encoding $false
        [System.IO.File]WriteAllText($viteFile, $newContent, $enc)
        C   vite.config patchato (allowedHosts + openfalse) Green
    } else {
        # Prova a cercare l'ultima } singola
        $lastBrace2 = $content.LastIndexOf(})
        if ($lastBrace2 -gt 0) {
            $newContent = $content.Substring(0, $lastBrace2) + $patch + }
            $enc = New-Object System.Text.UTF8Encoding $false
            [System.IO.File]WriteAllText($viteFile, $newContent, $enc)
            C   vite.config patchato (forma alternativa) Green
        } else {
            C   WARN vite.config non ha struttura riconosciuta, patch saltata Yellow
        }
    }
}

# ============================================================================
# AUTO-GENERA CADDYFILE per Tuttoinsieme
# ============================================================================
function Ensure-Caddyfile {
    param([string]$root, [int]$frontPort, [int]$backPort, [int]$proxyPort)

    $caddyPath = Join-Path $root Caddyfile

    if (Test-Path -LiteralPath $caddyPath) {
        # Verifica che il Caddyfile esistente usi la porta giusta
        $content = Get-Content -LiteralPath $caddyPath -Raw -ErrorAction SilentlyContinue
        if ($content -match 127.0.0.1$frontPort) {
            C   Caddyfile già corretto (skip) DarkCyan
            return
        }
        # Backup e ricrea
        Copy-Item -LiteralPath $caddyPath -Destination ($caddyPath + .BAK_PRIMA_PATCH) -Force
        C   Caddyfile aggiornato (porte corrette) Green
    } else {
        C   Caddyfile creato Green
    }

    $lines = @(
        http127.0.0.1$proxyPort {,
            # API e Sanctum - Laravel,
            handle api {,
                reverse_proxy 127.0.0.1$backPort,
            },
            handle sanctum {,
                reverse_proxy 127.0.0.1$backPort,
            },
            # Tutto il resto - Nuxt,
            handle {,
                reverse_proxy 127.0.0.1$frontPort,
            },
        }
    )
    Write-AsciiFile $caddyPath $lines
}

function Get-BPPanelLines {
    return @(

'$ErrorActionPreference = Stop',
'',
'$ScriptDir  = Split-Path -Parent $MyInvocation.MyCommand.Path',
'$Project    = BIANCHI PRO',
'$FrontDir   = $ScriptDir',
'$FrontPort  = 3000',
'$LocalUrl   = http127.0.0.1$FrontPort',
'$LogDir     = Join-Path $ScriptDir _LOGS',
'$StateFile  = Join-Path $ScriptDir _PANNELLO_STATE.json',
'$UrlFile    = Join-Path $ScriptDir URL_ONLINE.txt',
'',
'function Info([string]$m)  { Write-Host $m -ForegroundColor Cyan }',
'function Ok([string]$m)    { Write-Host $m -ForegroundColor Green }',
'function Warn([string]$m)  { Write-Host $m -ForegroundColor Yellow }',
'function Err([string]$m)   { Write-Host $m -ForegroundColor Red }',
'function Ensure-Dir([string]$p) { if(-not (Test-Path -LiteralPath $p)){ New-Item -ItemType Directory -Path $p -Force  Out-Null } }',
'function ReadKeyChar() { $k = $Host.UI.RawUI.ReadKey(NoEcho,IncludeKeyDown); return $k.Character.ToString().ToUpper() }',

'',
'function FindExe([string]$name) {',
'  $cmd = Get-Command $name -ErrorAction SilentlyContinue',
'  if ($cmd -and $cmd.Source) { return $cmd.Source }',
'  $local = Join-Path $ScriptDir $name',
'  if (Test-Path -LiteralPath $local) { return $local }',
'  $bin = Join-Path $envLOCALAPPDATA PANNELLO_BIN',
'  $binExe = Join-Path $bin $name',
'  if (Test-Path -LiteralPath $binExe) { return $binExe }',
'  return $null',
'}',
'',
'function TestTcpPort([int]$port, [int]$timeoutMs = 400) {',
'  try {',
'    $cl = New-Object System.Net.Sockets.TcpClient',
'    $iar = $cl.BeginConnect(127.0.0.1, $port, $null, $null)',
'    $ok  = $iar.AsyncWaitHandle.WaitOne($timeoutMs, $false)',
'    if (-not $ok) { $cl.Close(); return $false }',
'    $cl.EndConnect($iar)  Out-Null',
'    $cl.Close()',
'    return $true',
'  } catch { return $false }',
'}',
'function WaitPort([int]$port, [int]$timeoutSec) {',
'  $end = (Get-Date).AddSeconds($timeoutSec)',
'  while ((Get-Date) -lt $end) {',
'    if (TestTcpPort $port 400) { return $true }',
'    Start-Sleep -Milliseconds 300',
'  }',
'  return $false',
'}',
'',
'function LoadState() {',
'  if (Test-Path -LiteralPath $StateFile) {',
'    try { return (Get-Content -LiteralPath $StateFile -Raw  ConvertFrom-Json) } catch { return $null }',
'  }',
'  return $null',
'}',
'function SaveState($st) { ($st  ConvertTo-Json -Depth 6)  Set-Content -LiteralPath $StateFile -Encoding Ascii -Force }',
'function NewState() { return [PSCustomObject]@{ front_pid = 0; tunnel_pid = 0; last_url =  } }',

'    }',
'  } catch {}',

'',

'  Info CHIUSURA - $Project',
'  $st = LoadState',
'  if ($st) { KillTree ([int]$st.tunnel_pid); KillTree ([int]$st.front_pid) }',
'  KillByPort $FrontPort',
'  if (Test-Path -LiteralPath $StateFile) { Remove-Item -LiteralPath $StateFile -Force -ErrorAction SilentlyContinue }',
'  Ok OK chiuso.',
'}',
'',
'function Start-Local { param([bool]$OpenBrowser = $true)',
'  StopAll',
'  Ensure-Dir $LogDir',
'  $out = Join-Path $LogDir front_out.log',
'  $err = Join-Path $LogDir front_err.log',
'  foreach ($lf in @($out,$err)) { if (Test-Path -LiteralPath $lf) { Clear-Content -LiteralPath $lf -ErrorAction SilentlyContinue } }',
'',

'  $proc = Start-Process -FilePath cmd.exe -ArgumentList @(c,$cmdLine) -WorkingDirectory $FrontDir -WindowStyle Hidden -RedirectStandardOutput $out -RedirectStandardError $err -PassThru',
'',
'  $st = NewState; $st.front_pid = $proc.Id; SaveState $st',
'  Info (Attendo porta  + $FrontPort +  (max 90s)...)',
'  if (-not (WaitPort $FrontPort 90)) { Err Non pronto entro 90s. Aprire _LOGSfront_err.log; return $false }',
'  Ok (Locale pronto  + $LocalUrl)',
'  if ($OpenBrowser) { Start-Process $LocalUrl }',
'  return $true',
'}',
'',
'function WaitCloudUrl([string]$outPath, [string]$errPath, [int]$timeoutSec) {',
'  $end = (Get-Date).AddSeconds($timeoutSec)',
'  while ((Get-Date) -lt $end) {',
'    $txt = ',
'    if (Test-Path -LiteralPath $outPath) { $txt += (Get-Content -LiteralPath $outPath -Raw -ErrorAction SilentlyContinue) }',
'    if (Test-Path -LiteralPath $errPath) { $txt += (Get-Content -LiteralPath $errPath -Raw -ErrorAction SilentlyContinue) }',
'    $mtch = [regex]Match($txt, https[a-z0-9-]+.trycloudflare.com)',
'    if ($mtch.Success) { return $mtch.Value }',
'    Start-Sleep -Milliseconds 500',
'  }',
'  return ',
'}',
'',
'# Aspetta che il link Cloudflare risponda davvero a richieste HTTP (DNS + tunnel pronti)',
'function WaitCloudReady([string]$url, [int]$timeoutSec) {',
'  Info (Attendo che il tunnel sia raggiungibile  + $url)',
'  $end = (Get-Date).AddSeconds($timeoutSec)',
'  $attempt = 0',
'  while ((Get-Date) -lt $end) {',
'    $attempt++',
'    try {',
'      # Ignora errori SSL e redirect basta che risponda (qualsiasi HTTP status)',
'      $req = [System.Net.HttpWebRequest]Create($url)',
'      $req.Timeout = 5000',
'      $req.AllowAutoRedirect = $true',
'      $req.ServerCertificateValidationCallback = { $true }',
'      $resp = $req.GetResponse()',
'      $resp.Close()',
'      Ok (Tunnel pronto dopo  + $attempt +  tentativi.)',
'      return $true',
'    } catch [System.Net.WebException] {',
'      # 4xx5xx sono comunque risposte valide il tunnel funziona',
'      if ($_.Exception.Response -ne $null) {',
'        Ok (Tunnel pronto (HTTP  + [int]$_.Exception.Response.StatusCode + ).)',
'        return $true',
'      }',
'      # Altrimenti DNS non ancora pronto, attendi',
'      if ($attempt % 5 -eq 0) { Warn (  ancora in attesa... ( + $attempt +  tentativi)) }',
'      Start-Sleep -Milliseconds 1500',
'    } catch {',
'      Start-Sleep -Milliseconds 1500',
'    }',
'  }',
'  return $false',
'}',
'',
'function Share-Online {',
'  Ensure-Dir $LogDir',
'  if (-not (TestTcpPort $FrontPort 400)) {',
'    Warn Locale non attivo, avvio in corso...',
'    if (-not (Start-Local $false)) { return $false }',
'  }',
'',
'  $cloud = FindExe cloudflared.exe',
'  if (-not $cloud) { Err cloudflared.exe non trovato (PATH o cartella progetto).; return $false }',
'',
'  $st = LoadState; if (-not $st) { $st = NewState }',
'  if ([int]$st.tunnel_pid -gt 0) { KillTree ([int]$st.tunnel_pid) }',
'',
'  $tout = Join-Path $LogDir tunnel_out.log',
'  $terr = Join-Path $LogDir tunnel_err.log',
'  foreach ($lf in @($tout,$terr)) { if (Test-Path -LiteralPath $lf) { Clear-Content -LiteralPath $lf -ErrorAction SilentlyContinue } }',
'',
'  # NOTA NON usare $args (variabile riservata PowerShell) - usare $cloudArgs',
'  $cloudArgs = @(tunnel,--url,http127.0.0.1 + $FrontPort,--http-host-header,localhost + $FrontPort,--no-autoupdate)',
'  $tp = Start-Process -FilePath $cloud -ArgumentList $cloudArgs -WorkingDirectory $ScriptDir -WindowStyle Hidden -RedirectStandardOutput $tout -RedirectStandardError $terr -PassThru',
'  $st.tunnel_pid = $tp.Id; SaveState $st',
'',
'  Info Attendo URL dal log Cloudflare (max 60s)...',
'  $url = WaitCloudUrl $tout $terr 60',
'  if ($url -eq ) { Err URL non trovato nel log. Aprire _LOGStunnel_err.log; return $false }',
'  Info (URL trovato  + $url)',
'',
'  # Aspetta che il tunnel risponda davvero (DNS propagation) prima di aprire il browser',
'  if (-not (WaitCloudReady $url 90)) {',
'    Warn Timeout verifica HTTP. Apro il browser comunque (potrebbe servire un refresh).',
'  }',
'',
'  $st.last_url = $url; SaveState $st',
'  Set-Content -LiteralPath $UrlFile -Value $url -Encoding Ascii -Force',
'  Ok (Online pronto  + $url)',
'  Start-Process $url',
'  return $true',
'}',
'',
'function Open-Logs { Ensure-Dir $LogDir; Start-Process explorer.exe $LogDir }',
'',
'function Title {',
'  Clear-Host',
'  Write-Host ========================================== -ForegroundColor DarkCyan',
'  Write-Host (  PANNELLO -  + $Project) -ForegroundColor Cyan',
'  Write-Host ========================================== -ForegroundColor DarkCyan',
'  Write-Host ',
'}',
'',
'function Menu {',
'  while ($true) {',
'    Title',
'    Write-Host   [1] Avvia locale (apre UNA sola scheda) -ForegroundColor Green',
'    Write-Host   [2] Condividi online (apre SOLO link Cloudflare) -ForegroundColor Yellow',
'    Write-Host   [3] Apri cartella log -ForegroundColor Cyan',
'    Write-Host   [4] Chiudi tutto -ForegroundColor Red',
'    Write-Host   [Q] Esci -ForegroundColor Gray',
'    Write-Host ',
'    Write-Host   Premi un tasto (senza Invio)... -ForegroundColor DarkYellow',
'    $k = ReadKeyChar',
'    switch ($k) {',
'      1 { Start-Local $true   Out-Null; PauseKey }',
'      2 { Share-Online        Out-Null; PauseKey }',
'      3 { Open-Logs;                     PauseKey }',
'      4 { StopAll;                       PauseKey }',
'      Q { return }',
'    }',
'  }',
'}',
'',
'switch ($Action) {',
'  start { Start-Local $true   Out-Null; PauseKey }',
'  share { Share-Online        Out-Null; PauseKey }',
'  stop  { StopAll;                       PauseKey }',
'  logs  { Open-Logs;                     PauseKey }',
'  default { Menu }',
'}'
    )
}

'  if (-not (WaitPort $BackPort 90)) { Err Laravel non pronto. Aprire _LOGSback_err.log; return $false }',
'  Ok (Laravel pronto http127.0.0.1 + $BackPort)',
'',
'  # --- Frontend (Nuxt) ---',
'  Info (Avvio Nuxt su porta  + $FrontPort + ...)',
'  $fCmd = ''cd d '' + $FrontDir + '' && npm run dev''',
'  $fp = Start-Process -FilePath cmd.exe -ArgumentList @(c,$fCmd) -WorkingDirectory $FrontDir -WindowStyle Hidden -RedirectStandardOutput $fout -RedirectStandardError $ferr -PassThru',
'  $st.front_pid = $fp.Id; SaveState $st',
'  if (-not (WaitPort $FrontPort 120)) { Err Nuxt non pronto. Aprire _LOGSfront_err.log; return $false }',
'  Ok (Nuxt pronto http127.0.0.1 + $FrontPort)',
'',
'  # --- Proxy (Caddy) ---',
'  $caddy = FindExe caddy.exe',
'  if (-not $caddy) { Err caddy.exe non trovato (PATH o cartella progetto).; return $false }',
'',
'  Info (Avvio Caddy su porta  + $ProxyPort + ...)',
'  $cArgs = @(run,--config,$CaddyFile,--adapter,caddyfile)',
'  $cp = Start-Process -FilePath $caddy -ArgumentList $cArgs -WorkingDirectory $ScriptDir -WindowStyle Hidden -RedirectStandardOutput $cout -RedirectStandardError $cerr -PassThru',
'  $st.caddy_pid = $cp.Id; SaveState $st',
'  if (-not (WaitPort $ProxyPort 60)) { Err Caddy non pronto. Aprire _LOGScaddy_err.log; return $false }',
'  Ok (Locale pronto  + $LocalUrl)',
'  if ($OpenBrowser) { Start-Process $LocalUrl }',
'  return $true',
'}',
'',
'function WaitCloudUrl([string]$outPath, [string]$errPath, [int]$timeoutSec) {',
'  $end = (Get-Date).AddSeconds($timeoutSec)',
'  while ((Get-Date) -lt $end) {',
'    $txt = ',
'    if (Test-Path -LiteralPath $outPath) { $txt += (Get-Content -LiteralPath $outPath -Raw -ErrorAction SilentlyContinue) }',
'    if (Test-Path -LiteralPath $errPath) { $txt += (Get-Content -LiteralPath $errPath -Raw -ErrorAction SilentlyContinue) }',
'    $mtch = [regex]Match($txt, https[a-z0-9-]+.trycloudflare.com)',
'    if ($mtch.Success) { return $mtch.Value }',
'    Start-Sleep -Milliseconds 500',
'  }',
'  return ',
'}',
'',
'# Aspetta che il link Cloudflare risponda davvero a richieste HTTP (DNS + tunnel pronti)',
'function WaitCloudReady([string]$url, [int]$timeoutSec) {',
'  Info (Attendo che il tunnel sia raggiungibile  + $url)',
'  $end = (Get-Date).AddSeconds($timeoutSec)',
'  $attempt = 0',
'  while ((Get-Date) -lt $end) {',
'    $attempt++',
'    try {',
'      $req = [System.Net.HttpWebRequest]Create($url)',
'      $req.Timeout = 5000',
'      $req.AllowAutoRedirect = $true',
'      $req.ServerCertificateValidationCallback = { $true }',
'      $resp = $req.GetResponse()',
'      $resp.Close()',
'      Ok (Tunnel pronto dopo  + $attempt +  tentativi.)',
'      return $true',
'    } catch [System.Net.WebException] {',
'      if ($_.Exception.Response -ne $null) {',
'        Ok (Tunnel pronto (HTTP  + [int]$_.Exception.Response.StatusCode + ).)',
'        return $true',
'      }',
'      if ($attempt % 5 -eq 0) { Warn (  ancora in attesa... ( + $attempt +  tentativi)) }',
'      Start-Sleep -Milliseconds 1500',
'    } catch {',
'      Start-Sleep -Milliseconds 1500',
'    }',
'  }',
'  return $false',
'}',
'',
'function Share-Online {',
'  Ensure-Dir $LogDir',
'  if (-not (TestTcpPort $ProxyPort 400)) {',
'    Warn Locale non attivo, avvio in corso...',
'    if (-not (Start-Local $false)) { return $false }',
'  }',
'',
'  $cloud = FindExe cloudflared.exe',
'  if (-not $cloud) { Err cloudflared.exe non trovato (PATH o cartella progetto).; return $false }',
'',
'  $st = LoadState; if (-not $st) { $st = NewState }',
'  if ([int]$st.tunnel_pid -gt 0) { KillTree ([int]$st.tunnel_pid) }',
'',
'  $tout = Join-Path $LogDir tunnel_out.log',
'  $terr = Join-Path $LogDir tunnel_err.log',
'  foreach ($lf in @($tout,$terr)) { if (Test-Path -LiteralPath $lf) { Clear-Content -LiteralPath $lf -ErrorAction SilentlyContinue } }',
'',
'  # NOTA NON usare $args (variabile riservata PowerShell) - usare $cloudArgs',
'  $cloudArgs = @(tunnel,--url,http127.0.0.1 + $ProxyPort,--http-host-header,localhost + $ProxyPort,--no-autoupdate)',
'  $tp = Start-Process -FilePath $cloud -ArgumentList $cloudArgs -WorkingDirectory $ScriptDir -WindowStyle Hidden -RedirectStandardOutput $tout -RedirectStandardError $terr -PassThru',
'  $st.tunnel_pid = $tp.Id; SaveState $st',
'',
'  Info Attendo URL dal log Cloudflare (max 60s)...',
'  $url = WaitCloudUrl $tout $terr 60',
'  if ($url -eq ) { Err URL non trovato nel log. Aprire _LOGStunnel_err.log; return $false }',
'  Info (URL trovato  + $url)',
'',
'  # Aspetta che il tunnel risponda davvero (DNS propagation) prima di aprire il browser',
'  if (-not (WaitCloudReady $url 90)) {',
'    Warn Timeout verifica HTTP. Apro il browser comunque (potrebbe servire un refresh).',
'  }',
'',
'  $st.last_url = $url; SaveState $st',
'  Set-Content -LiteralPath $UrlFile -Value $url -Encoding Ascii -Force',
'  Ok (Online pronto  + $url)',
'  Start-Process $url',
'  return $true',
'}',
'',
'function Open-Logs { Ensure-Dir $LogDir; Start-Process explorer.exe $LogDir }',
'',
'function Title {',
'  Clear-Host',
'  Write-Host ========================================== -ForegroundColor DarkCyan',
'  Write-Host (  PANNELLO -  + $Project) -ForegroundColor Cyan',
'  Write-Host ========================================== -ForegroundColor DarkCyan',
'  Write-Host ',
'}',
'',
'function Menu {',
'  while ($true) {',
'    Title',
'    Write-Host   [1] Avvia locale (apre UNA sola scheda) -ForegroundColor Green',
'    Write-Host   [2] Condividi online (apre SOLO link Cloudflare) -ForegroundColor Yellow',
'    Write-Host   [3] Apri cartella log -ForegroundColor Cyan',
'    Write-Host   [4] Chiudi tutto -ForegroundColor Red',
'    Write-Host   [Q] Esci -ForegroundColor Gray',
'    Write-Host ',
'    Write-Host   Premi un tasto (senza Invio)... -ForegroundColor DarkYellow',
'    $k = ReadKeyChar',
'    switch ($k) {',
'      1 { Start-Local $true   Out-Null; PauseKey }',
'      2 { Share-Online        Out-Null; PauseKey }',
'      3 { Open-Logs;                     PauseKey }',
'      4 { StopAll;                       PauseKey }',
'      Q { return }',
'    }',
'  }',
'}',
'',
'switch ($Action) {',
'  start { Start-Local $true   Out-Null; PauseKey }',
'  share { Share-Online        Out-Null; PauseKey }',
'  stop  { StopAll;                       PauseKey }',
'  logs  { Open-Logs;                     PauseKey }',
'  default { Menu }',
'}'
    )
}

$desktop = Find-Desktop
C (Desktop  + $desktop) DarkCyan

# Configurazione progetti
$projects = @(
    @{
        Root     = Join-Path $desktop BianchiproRestyling
        Kind     = BP
        Name     = BianchiproRestyling
        NuxtDir  = $null
        ViteDir  = Join-Path $desktop BianchiproRestyling
    },
    @{
        Root     = Join-Path $desktop tuttoinsieme
        Kind     = TI
        Name     = tuttoinsieme
        NuxtDir  = Join-Path (Join-Path $desktop tuttoinsieme) nuxt-spedizionefacile-master
        ViteDir  = $null
    }
)

foreach ($proj in $projects) {
    C 
    C (---  + $proj.Name +  ---) Cyan

    if (-not (Test-Path -LiteralPath $proj.Root)) {
        C (  SKIP cartella non trovata  + $proj.Root) Yellow
        continue
    }

    # 1. Pulizia file vecchi
    Cleanup-Old $proj.Root
    Ensure-Dir (Join-Path $proj.Root _LOGS)

    # 2. Patch config automatici
    if ($proj.Kind -eq BP) {
        Patch-ViteConfig -projDir $proj.ViteDir
    }
    if ($proj.Kind -eq TI) {
        if ($proj.NuxtDir -and (Test-Path -LiteralPath $proj.NuxtDir)) {
            Patch-NuxtConfig -nuxtDir $proj.NuxtDir -port 3001
        } else {
            C (  WARN cartella Nuxt non trovata  + $proj.NuxtDir) Yellow
        }
        # 3. Auto-genera Caddyfile
        Ensure-Caddyfile -root $proj.Root -frontPort 3001 -backPort 8000 -proxyPort 8787
    }

    # 4. Crea i .bat di scorciatoia
    $batPanel  = @('@echo off','cd d %~dp0','powershell -NoProfile -ExecutionPolicy Bypass -File %~dp0PANNELLO.ps1')
    $batStart  = @('@echo off','cd d %~dp0','powershell -NoProfile -ExecutionPolicy Bypass -File %~dp0PANNELLO.ps1 start','pause nul')
    $batShare  = @('@echo off','cd d %~dp0','powershell -NoProfile -ExecutionPolicy Bypass -File %~dp0PANNELLO.ps1 share','pause nul')
    $batStop   = @('@echo off','cd d %~dp0','powershell -NoProfile -ExecutionPolicy Bypass -File %~dp0PANNELLO.ps1 stop','pause nul')

    Write-AsciiFile (Join-Path $proj.Root PANNELLO.bat)        $batPanel
    Write-AsciiFile (Join-Path $proj.Root AVVIA_LOCALE.bat)    $batStart
    Write-AsciiFile (Join-Path $proj.Root CONDIVIDI_ONLINE.bat) $batShare
    Write-AsciiFile (Join-Path $proj.Root CHIUDI_TUTTO.bat)    $batStop

    # 5. Scrive il PANNELLO.ps1 corretto
    if ($proj.Kind -eq BP) {
        Write-AsciiFile (Join-Path $proj.Root PANNELLO.ps1) (Get-BPPanelLines)
    } else {
        Write-AsciiFile (Join-Path $proj.Root PANNELLO.ps1) (Get-TIPanelLines)
    }

    C (  OK tutto installato in  + $proj.Root) Green
}
